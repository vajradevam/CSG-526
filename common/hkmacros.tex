% Shared TikZ styles, colours and the running example.
% Used by both slides/slides.tex and notes/lecture.tex.
%
% Running example (X = {a,...,e}, Y = {1,...,5}):
%   edges  a1 a3 | b2 b5 | c4 c5 | d1 d2 | e1
%   Phase 1: shortest aug. path length 1  -> a1, b2, c4
%   Phase 2: shortest aug. path length 3  -> d1 a3
%   Phase 3: shortest aug. path length 5  -> e1 d2 b5   (perfect matching)

\usepackage{amsmath,amssymb,mathtools}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds,decorations.pathreplacing,shapes.geometric}

\definecolor{hkblue}{HTML}{2F5D9B}
\definecolor{hkorange}{HTML}{D1793A}
\definecolor{hkred}{HTML}{C0392B}
\definecolor{hkgreen}{HTML}{2E8B57}
\definecolor{hkgrey}{HTML}{6B6B6B}

\newcommand{\symdiff}{\oplus}
\newcommand{\M}{M}
\newcommand{\bigO}[1]{O\!\left(#1\right)}

\tikzset{
  boy/.style={circle, draw=hkblue!80, fill=hkblue!16, minimum size=6.6mm,
              inner sep=0pt, font=\footnotesize, thick},
  girl/.style={circle, draw=hkorange!90, fill=hkorange!22, minimum size=6.6mm,
               inner sep=0pt, font=\footnotesize, thick},
  freev/.style={draw=hkgreen, line width=1.3pt},   % ring that marks a free vertex
  unm/.style={line width=0.6pt, draw=black!35},          % unmatched edge
  mat/.style={line width=1.9pt, draw=black!80},          % matching edge
  aug/.style={line width=2.2pt, draw=hkred},             % edge on augmenting path
  Medge/.style={line width=1.9pt, draw=hkblue},
  Nedge/.style={line width=1.9pt, draw=hkorange, dashed},
  layerlab/.style={font=\footnotesize\bfseries, text=hkblue!85!black},
}

% ---- the running example --------------------------------------------------
% left column top->bottom: a e d b c ; right column: 3 1 2 5 4   (planar drawing)
\newcommand{\exnodes}{%
  \node[boy]  (xa) at (0, 2.2) {$a$};
  \node[boy]  (xe) at (0, 1.1) {$e$};
  \node[boy]  (xd) at (0, 0.0) {$d$};
  \node[boy]  (xb) at (0,-1.1) {$b$};
  \node[boy]  (xc) at (0,-2.2) {$c$};
  \node[girl] (y3) at (2.8, 2.2) {$3$};
  \node[girl] (y1) at (2.8, 1.1) {$1$};
  \node[girl] (y2) at (2.8, 0.0) {$2$};
  \node[girl] (y5) at (2.8,-1.1) {$5$};
  \node[girl] (y4) at (2.8,-2.2) {$4$};
}
\newcommand{\exedges}{%
  \draw[unm] (xa)--(y1); \draw[unm] (xa)--(y3);
  \draw[unm] (xb)--(y2); \draw[unm] (xb)--(y5);
  \draw[unm] (xc)--(y4); \draw[unm] (xc)--(y5);
  \draw[unm] (xd)--(y1); \draw[unm] (xd)--(y2);
  \draw[unm] (xe)--(y1);
}
% highlight a vertex as free
\newcommand{\freering}[1]{\draw[freev] (#1) circle (4.4mm);}

% ---- one phase at a glance: edge styles, graph states, level pictures --------
\tikzset{lnon/.style={line width=0.9pt,draw=black!45},
        lmat/.style={line width=2pt,draw=black!80},
        lred/.style={line width=2.4pt,draw=hkred}}
\newcommand{\usedv}[1]{\fill[white,opacity=0.6] (#1) circle (4.3mm);}
\newcommand{\deadx}[1]{\draw[hkred,thick] ($(#1)+(-0.28,-0.28)$)--($(#1)+(0.28,0.28)$) ($(#1)+(-0.28,0.28)$)--($(#1)+(0.28,-0.28)$);}

% graph states of the running example (planar drawing); green rings = free vertices
\newcommand{\exStateA}{\exnodes\exedges % M = empty
  \freering{xa}\freering{xb}\freering{xc}\freering{xd}\freering{xe}\freering{y1}\freering{y2}\freering{y3}\freering{y4}\freering{y5}}
\newcommand{\exStateB}{\exnodes\exedges % M = {a1,b2,c4}
  \draw[mat] (xa)--(y1); \draw[mat] (xb)--(y2); \draw[mat] (xc)--(y4);
  \freering{xd}\freering{xe}\freering{y3}\freering{y5}}
\newcommand{\exStateC}{\exnodes\exedges % M = {d1,a3,b2,c4}
  \draw[mat] (xd)--(y1); \draw[mat] (xa)--(y3); \draw[mat] (xb)--(y2); \draw[mat] (xc)--(y4);
  \freering{xe}\freering{y5}}
\newcommand{\exStateD}{\exnodes\exedges % M = {e1,d2,b5,a3,c4}: perfect
  \draw[mat] (xe)--(y1); \draw[mat] (xd)--(y2); \draw[mat] (xb)--(y5); \draw[mat] (xa)--(y3); \draw[mat] (xc)--(y4);}

% level pictures, drawn top-down (level 0 on top), units in cm
% phase 1 (M empty): two levels
\newcommand{\phOneNodes}{%
  \node[boy] (Qa) at (-1.8,0) {$a$}; \node[boy] (Qe) at (-0.9,0) {$e$}; \node[boy] (Qd) at (0,0) {$d$};
  \node[boy] (Qb) at (0.9,0) {$b$}; \node[boy] (Qc) at (1.8,0) {$c$};
  \node[girl] (Q3) at (-1.8,-2) {$3$}; \node[girl] (Q1) at (-0.9,-2) {$1$}; \node[girl] (Q2) at (0,-2) {$2$};
  \node[girl] (Q5) at (0.9,-2) {$5$}; \node[girl] (Q4) at (1.8,-2) {$4$};
  \node[layerlab,anchor=east] at (-2.3,0) {level 0}; \node[layerlab,anchor=east] at (-2.3,-2) {level 1};}
\newcommand{\phOneEdges}{%
  \draw[lnon] (Qa)--(Q1); \draw[lnon] (Qa)--(Q3); \draw[lnon] (Qb)--(Q2); \draw[lnon] (Qb)--(Q5);
  \draw[lnon] (Qc)--(Q4); \draw[lnon] (Qc)--(Q5); \draw[lnon] (Qd)--(Q1); \draw[lnon] (Qd)--(Q2); \draw[lnon] (Qe)--(Q1);}
\newcommand{\phOneDFS}{%
  \draw[lred] (Qa)--(Q1); \draw[lred] (Qb)--(Q2); \draw[lred] (Qc)--(Q4);
  \usedv{Qa}\usedv{Q1}\usedv{Qb}\usedv{Q2}\usedv{Qc}\usedv{Q4}\deadx{Q3}\deadx{Q5}}

% phase 2 (M = {a1,b2,c4}): four levels
\newcommand{\phTwoNodes}{%
  \node[boy] (Pd) at (-0.9,0) {$d$}; \node[boy] (Pe) at (0.9,0) {$e$};
  \node[girl] (P1) at (-0.9,-1.2) {$1$}; \node[girl] (P2) at (0.9,-1.2) {$2$};
  \node[boy] (Pa) at (-0.9,-2.4) {$a$}; \node[boy] (Pb) at (0.9,-2.4) {$b$};
  \node[girl] (P3) at (-0.9,-3.6) {$3$}; \node[girl] (P5) at (0.9,-3.6) {$5$};
  \foreach \i in {0,1,2,3} \node[layerlab,anchor=east] at (-1.5,-1.2*\i) {level \i};
  \freering{Pd}\freering{Pe}\freering{P3}\freering{P5}}
\newcommand{\phTwoEdges}{%
  \draw[lnon] (Pd)--(P1); \draw[lnon] (Pe)--(P1); \draw[lnon] (Pd)--(P2);
  \draw[lmat] (P1)--(Pa); \draw[lmat] (P2)--(Pb);
  \draw[lnon] (Pa)--(P3); \draw[lnon] (Pb)--(P5);}
\newcommand{\phTwoDFS}{%
  \draw[lred] (Pd)--(P1); \draw[lred] (P1)--(Pa); \draw[lred] (Pa)--(P3);
  \usedv{Pd}\usedv{P1}\usedv{Pa}\usedv{P3}\deadx{P5}\deadx{Pb}\deadx{P2}}

% phase 3 (M = {d1,a3,b2,c4}): a single chain of six levels
\newcommand{\phThreeNodes}{%
  \node[boy] (Se) at (0,0) {$e$}; \node[girl] (S1) at (0,-0.85) {$1$}; \node[boy] (Sd) at (0,-1.7) {$d$};
  \node[girl] (S2) at (0,-2.55) {$2$}; \node[boy] (Sb) at (0,-3.4) {$b$}; \node[girl] (S5) at (0,-4.25) {$5$};
  \foreach \i in {0,1,2,3,4,5} \node[layerlab,anchor=east] at (-0.6,-0.85*\i) {level \i};
  \freering{Se}\freering{S5}}
\newcommand{\phThreeEdges}{%
  \draw[lnon] (Se)--(S1); \draw[lmat] (S1)--(Sd); \draw[lnon] (Sd)--(S2);
  \draw[lmat] (S2)--(Sb); \draw[lnon] (Sb)--(S5);}
\newcommand{\phThreeDFS}{%
  \draw[lred] (Se)--(S1); \draw[lred] (S1)--(Sd); \draw[lred] (Sd)--(S2);
  \draw[lred] (S2)--(Sb); \draw[lred] (Sb)--(S5);}
